cmake_minimum_required(VERSION 3.24)
project(starz_s1 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(STARZ_BUILD_ENGINEERING "Build parts-first S1-B CAD and numerical engineering" ON)
option(STARZ_BUILD_HMI "Build the native Qt desktop application" ON)
option(STARZ_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(STARZ_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_subdirectory(software)
if(STARZ_BUILD_ENGINEERING)
  add_subdirectory(engineering)
endif()
add_test(NAME production_record_integrity
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/check_production_records.py")
add_test(NAME check_design_integrity
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/check_design_integrity.py")
add_test(NAME if_b01_rp2350_gpio_allocation
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/check_if_b01_gpio_mapping.py")
add_test(NAME mechanism_generated_output_integrity
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/check_mechanism_outputs.py")
